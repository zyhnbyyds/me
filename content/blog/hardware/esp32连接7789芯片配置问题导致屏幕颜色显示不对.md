## 1. 背景

最近在学习硬件，连接屏幕显示模块的时候，遇到一个问题，就是屏幕显示的颜色不对，经过排查，发现是屏幕的 **esp_lcd_panel_dev_config_t** 中的 **data_endian** 配置有问题，导致屏幕显示的颜色不对。

## 2. 硬件参数

- **主控** ESP32-s3-N16R8
- **屏幕模块** ST7789V2(240 * 280)

## 3. 该配置作用

所以最后颜色显示不对的原因找到了，默认这个配置是 **LCD_RGB_DATA_ENDIAN_BIG**，也就是默认从大端序列发送数据，将这个改为**LCD_RGB_DATA_ENDIAN_LITTLE**后，颜色从小端读取，之前默认配置的颜色显示不对，字体虚化等问题都解决了。

## 4. 我的代码（简单的lvgl移植展示）

```c
void screen_mod_init()
{
  gpio_config_t bk_gpio_config = {
      .mode = GPIO_MODE_OUTPUT,
      .pin_bit_mask = 1ULL << EXAMPLE_PIN_NUM_BK_LIGHT};
  // Initialize the GPIO of backlight
  ESP_ERROR_CHECK(gpio_config(&bk_gpio_config));

  spi_bus_config_t buscfg = {
      .sclk_io_num = EXAMPLE_PIN_NUM_PCLK,
      .mosi_io_num = EXAMPLE_PIN_NUM_DATA0,
      .miso_io_num = -1,
      .quadwp_io_num = -1,
      .quadhd_io_num = -1,
      .max_transfer_sz = PARALLEL_LINES * HOR_RES * 2 + 8};

  // Initialize the SPI bus
  ESP_ERROR_CHECK(spi_bus_initialize(LCD_HOST, &buscfg, SPI_DMA_CH_AUTO));

  esp_lcd_panel_io_handle_t io_handle = NULL;
  esp_lcd_panel_io_spi_config_t io_config = {
      .dc_gpio_num = EXAMPLE_PIN_NUM_DC,
      .cs_gpio_num = EXAMPLE_PIN_NUM_CS,
      .pclk_hz = EXAMPLE_LCD_PIXEL_CLOCK_HZ,
      .lcd_cmd_bits = EXAMPLE_LCD_CMD_BITS,
      .lcd_param_bits = EXAMPLE_LCD_PARAM_BITS,
      .spi_mode = 0,
      .trans_queue_depth = 10,
  };

  // Attach the LCD to the SPI bus
  ESP_ERROR_CHECK(esp_lcd_new_panel_io_spi((esp_lcd_spi_bus_handle_t)LCD_HOST, &io_config, &io_handle));

  esp_lcd_panel_dev_config_t panel_config = {
      .reset_gpio_num = EXAMPLE_PIN_NUM_RST,
      .rgb_ele_order = LCD_RGB_ELEMENT_ORDER_BGR,
      .data_endian = LCD_RGB_DATA_ENDIAN_LITTLE,
      .bits_per_pixel = 16,
  };
  ESP_ERROR_CHECK(esp_lcd_new_panel_st7789(io_handle, &panel_config, &panel_handle));
  ESP_ERROR_CHECK(gpio_set_level(EXAMPLE_PIN_NUM_BK_LIGHT, EXAMPLE_LCD_BK_LIGHT_OFF_LEVEL));
  ESP_ERROR_CHECK(esp_lcd_panel_reset(panel_handle));
  ESP_ERROR_CHECK(esp_lcd_panel_init(panel_handle));
  ESP_ERROR_CHECK(esp_lcd_panel_disp_on_off(panel_handle, true));
  ESP_ERROR_CHECK(esp_lcd_panel_invert_color(panel_handle, true));
  // ESP_ERROR_CHECK(esp_lcd_panel_swap_xy(panel_handle, true));
  esp_lcd_panel_mirror(panel_handle, true, true);
  ESP_ERROR_CHECK(gpio_set_level(EXAMPLE_PIN_NUM_BK_LIGHT, EXAMPLE_LCD_BK_LIGHT_ON_LEVEL));

  lv_init();

  lv_display_t *disp = lv_display_create(HOR_RES, VER_RES);
  lv_display_set_flush_cb(disp, my_flush_cb);
  lv_display_set_buffers(disp, buf1, buf2, BUFFER_PIXEL_COUNT, mode);

  const esp_timer_create_args_t periodic_timer_args = {
      .callback = &lv_tick_task,
      .name = "lv_tick"};
  esp_timer_handle_t periodic_timer;
  esp_timer_create(&periodic_timer_args, &periodic_timer);
  esp_timer_start_periodic(periodic_timer, 1000);

  setup_ui(&guider_ui);

  xTaskCreatePinnedToCore(lvgl_timer_task, "lvgl", 4096, NULL, 5, NULL, 1);
  // xTaskCreatePinnedToCore(dht11_sceen_task, "dht11_screen", 4096, NULL, 4, NULL, 1);
}
```
